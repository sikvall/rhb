\section{Ändringar sedan förra utgåvan}
\index{ändringar}
\index{upplaga}

\begin{itemize}%[label={\\--}]
        \item Flygfrekvenser, VHF inom landet och HF över Atlanten.

	\item DTMF-toner, avsnitt \ref{DTMF}.

	\item Marin VHF: kanalplanen från 2022 samt VDES / VDE-SAT nu uppdaterad.

	\item Vågutbredning omarbetad, kapitel \ref{vågutbredning}:
	      jonosfärens skikt, flera hopp, fädning, NVIS, sporadiskt E,
	      tropo, meteorscatter och greyline.

	    \item Rymdväder utbyggt: solfackla, PCA, CME, koronahål, norrsken
	      och bandval. Hör ihop med Rymdväderappen.

	    \item Antenner att bygga själv: dipol, doublet,
              off-center, vertikal, ändmatad halvvåg, yagi,
              polarisation, VHF/70\,cm, portabelt samt matning och
              anpassning.

	    \item Radioberäkningar för VHF/UHF: Fresnelzon, tvåstråleutbredning,
	      länkbudget med exempel, avstånd och bäring ur lokator,
	      antennlängd.

	    \item Webbappar: Rymdväder (\ref{rymdväderapp}), greyline/termin
	      (\ref{termin}), Tempus, Tut, Remsan och QSO-logg.

	    \item Instegscert informationen uppdaterad

	    \item SWEN i testfas med kommersiella operatörer. RAKEL planeras
	      avvecklas 2030. Ansvarig myndighet är MCF.
\end{itemize}
